
CREATE PROCEDURE ApplyForJob
	@UserID INT,
	@JobID INT
AS
BEGIN
	SET NOCOUNT ON;

	IF NOT EXISTS(
		SELECT 1
		FROM Jobs
		WHERE JobID=@JobID
		AND Deadline > SYSUTCDATETIME()
	)
	BEGIN
		THROW 50001, 'Job does not exist or its deadline has expired.', 1;
	END;

	INSERT INTO Applications(UserID,JobID)
	VALUES(@UserID,@JobID);
END;
GO





CREATE FUNCTION GetApplicationCount
(
	@JobID INT
)
RETURNS INT
AS
BEGIN
	DECLARE @Count INT;

	SELECT @Count= COUNT(*)
	FROM Applications
	WHERE JobID=@JobID;

	RETURN @Count;
END;
GO







CREATE TRIGGER trg_PreventLateApplications
ON Applications
AFTER INSERT
AS
BEGIN
	SET NOCOUNT ON;

	IF EXISTS(
		SELECT 1
		FROM inserted AS i
		INNER JOIN Jobs AS j
			ON j.JobID=i.JobID
		WHERE j.Deadline <= SYSUTCDATETIME()
	)
	BEGIN
		THROW 50002, 'Cannot apply after the job deadline.', 1;
    END;
END;
GO
